Для начало написания кода нужно импортировать следующее

from aiogram import Router
from aiogram.types import Message
from aiogram.filters import Command

Остальное по желанию

Добавляйте строго роутер

router = Router()

Вы без роутера не запустите скрипт

Скрипт со /start ОСТАВЛЯЕТЕ ТОЛЬКО ОДИН ИЛИ ПРОСИТЕ/ПЕРЕПИСЫВАЙТЕ МОДУЛЬ КОМАНДЫ ТОЖЕ ДОЛЖНЫ БЫТЬ ОДНИ

ПРИМЕР КОМАНД

@router.message(Command("start"))
async def cmd_start(message: Message):
    await message.answer(
        "Привет! Я пример модульного Telegram‑бота.\n"
        "Используйте /help для получения справки."
    )
	
ИЛИ
 
@router.message(Command("help"))
async def cmd_help(message: Message):
    help_text = (
        "📖 **Доступные команды:**\n\n"
        "/start — запустить бота\n"
        "/help — показать эту справку\n"
        "/info — получить информацию о боте\n"
        "/echo [текст] — повторить ваш текст"
    )
    await message.answer(help_text)
